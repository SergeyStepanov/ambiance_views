cmake_minimum_required(VERSION 3.20)
project(AmbianceViews VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TERRARIUM_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(SDL3 3.2 REQUIRED CONFIG)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(TERRARIUM_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(terrarium_core src/world.c)
target_include_directories(terrarium_core PUBLIC src)
if(UNIX)
  target_link_libraries(terrarium_core PUBLIC m)
endif()
enable_testing()
add_executable(test_simulation tests/test_simulation.c)
target_link_libraries(test_simulation PRIVATE terrarium_core)
add_test(NAME simulation COMMAND test_simulation)
add_library(terrarium_observation src/observation.c)
target_link_libraries(terrarium_observation PUBLIC terrarium_core)
add_library(terrarium_view src/render.c src/assets.c)
target_link_libraries(terrarium_view PUBLIC terrarium_observation SDL3::SDL3)
add_executable(ambiance_views src/main.c)
target_link_libraries(ambiance_views PRIVATE terrarium_view)
add_executable(test_observation tests/test_observation.c)
target_link_libraries(test_observation PRIVATE terrarium_observation)
add_test(NAME observation COMMAND test_observation)
add_executable(test_render tests/test_render.c)
target_link_libraries(test_render PRIVATE terrarium_view)
add_test(NAME render COMMAND test_render)
add_test(NAME headless_app COMMAND ambiance_views --headless --seed 42 --frames 30 --snapshot app-smoke.bmp)
add_executable(test_native tests/test_native.c)
target_link_libraries(test_native PRIVATE terrarium_view)
option(TERRARIUM_NATIVE_TESTS "Include native window checks in CTest (requires a display)" OFF)
if(TERRARIUM_NATIVE_TESTS)
  add_test(NAME native_window COMMAND test_native)
endif()
install(TARGETS ambiance_views RUNTIME DESTINATION bin)
